        // The installed, hash-pinned app itself exercises the XPC connection.
        // This path exits before monitoring or UI control loops can start.
        let localArguments = ProcessInfo.processInfo.arguments
        if localArguments.contains("--local-helper-status") || localArguments.contains("--local-self-test") {
            let performWrites = localArguments.contains("--local-self-test")
            var targets: [Int] = []
            if performWrites {
                let monitor = SystemMonitor()
                guard monitor.checkAccess() else { print("LOCAL_SELF_TEST=no-smc-access"); exit(2) }
                for fan in 0..<8 {
                    guard monitor.readSMCValue(key: "F\(fan)Ac") != nil else { break }
                    guard let maximum = monitor.readSMCValue(key: "F\(fan)Mx"), maximum > 0 else { exit(2) }
                    // Match the helper's integer conversion of the hardware limit.
                    targets.append(Int(maximum))
                }
                guard !targets.isEmpty else { exit(2) }
            }
            let maximumTargets = targets
            DispatchQueue.global(qos: .userInitiated).async {
                guard let state = SMCDaemonClient.pingState() else { print("LOCAL_HELPER_STATUS=unavailable"); exit(1) }
                print("LOCAL_HELPER_STATUS=\(state.rawValue)")
                if performWrites {
                    var manualOK = true
                    for (fan, maximum) in maximumTargets.enumerated() {
                        let reply = SMCDaemonClient.localCommandResponse("SETV2 \(fan) \(maximum)")
                        print("LOCAL_SET fan=\(fan) rpm=\(maximum) reply=\(reply ?? "no-reply")")
                        if reply != "OK" { manualOK = false }
                    }
                    if manualOK {
                        var observed: [[Int]] = []
                        var verified = false
                        for _ in 0..<25 {
                            observed = DispatchQueue.main.sync {
                                let monitor = SystemMonitor()
                                return maximumTargets.indices.map { fan in
                                    ["Md", "Tg", "Ac"].map { suffix in
                                        Int(monitor.readSMCValue(key: "F\(fan)\(suffix)") ?? -1)
                                    }
                                }
                            }
                            verified = observed.indices.allSatisfy { fan in
                                observed[fan][0] == 1 && abs(observed[fan][1] - maximumTargets[fan]) <= 1 && observed[fan][2] > 0
                            }
                            if verified { break }
                            Thread.sleep(forTimeInterval: 0.2)
                        }
                        print("LOCAL_MANUAL_READBACK [mode,target,rpm]=\(observed)")
                        manualOK = verified
                    }
                    // Always restore, including when SET or physical verification failed.
                    var systemOK = true
                    for fan in maximumTargets.indices {
                        let reply = SMCDaemonClient.localCommandResponse("AUTOV2 \(fan)")
                        print("LOCAL_AUTO fan=\(fan) reply=\(reply ?? "no-reply")")
                        if reply != "OK" { systemOK = false }
                    }
                    let systemModes: [Int] = DispatchQueue.main.sync {
                        let monitor = SystemMonitor()
                        return maximumTargets.indices.map { fan in
                            Int(monitor.readSMCValue(key: "F\(fan)Md") ?? -1)
                        }
                    }
                    print("LOCAL_SYSTEM_MODES=\(systemModes)")
                    systemOK = systemOK && systemModes.allSatisfy { $0 == 0 || $0 == 3 }
                    print("LOCAL_SELF_TEST manual=\(manualOK) system=\(systemOK)")
                    fflush(stdout)
                    exit(manualOK && systemOK ? 0 : 1)
                }
                fflush(stdout)
                exit(0)
            }
            return
        }
